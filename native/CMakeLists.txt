cmake_minimum_required(VERSION 3.21)
project(bitfroest_native LANGUAGES C CXX)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_BINARY_DIR)
  message(FATAL_ERROR "Use an external build directory.")
endif()
if(NOT EXISTS "${NODE_INCLUDE_DIR}/node_api.h")
  message(FATAL_ERROR "NODE_INCLUDE_DIR must contain the preinstalled Node-API headers.")
endif()
set(LIBGIT2_SOURCE_DIR "" CACHE PATH "Path to the verified libgit2 source tree.")
set(LIBGIT2_EXPECTED_VERSION "1.9.7")
set(LIBGIT2_EXPECTED_COMMIT "49e408b3208bc3093757a1c2db938d3590f3f412")
set(LIBGIT2_EXPECTED_TREE "29b18aa9210376513c391185b355e696856f9261")
if(NOT EXISTS "${LIBGIT2_SOURCE_DIR}/CMakeLists.txt")
  message(FATAL_ERROR "LIBGIT2_SOURCE_DIR must name the pinned libgit2 source tree.")
endif()
file(READ "${LIBGIT2_SOURCE_DIR}/CMakeLists.txt" LIBGIT2_TOP_CMAKE LIMIT 512)
string(FIND "${LIBGIT2_TOP_CMAKE}" "project(libgit2 VERSION \"${LIBGIT2_EXPECTED_VERSION}\"" LIBGIT2_VERSION_POSITION)
if(LIBGIT2_VERSION_POSITION EQUAL -1)
  message(FATAL_ERROR "The libgit2 source version does not match the required version.")
endif()
find_package(Git REQUIRED)
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LIBGIT2_SOURCE_DIR}" rev-parse HEAD
  OUTPUT_VARIABLE LIBGIT2_SOURCE_COMMIT OUTPUT_STRIP_TRAILING_WHITESPACE
  RESULT_VARIABLE LIBGIT2_GIT_RESULT)
if(NOT LIBGIT2_GIT_RESULT EQUAL 0 OR NOT LIBGIT2_SOURCE_COMMIT STREQUAL LIBGIT2_EXPECTED_COMMIT)
  message(FATAL_ERROR "The libgit2 source commit does not match the required commit.")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LIBGIT2_SOURCE_DIR}" rev-parse "HEAD^{tree}"
  OUTPUT_VARIABLE LIBGIT2_SOURCE_TREE OUTPUT_STRIP_TRAILING_WHITESPACE
  RESULT_VARIABLE LIBGIT2_TREE_RESULT)
if(NOT LIBGIT2_TREE_RESULT EQUAL 0 OR NOT LIBGIT2_SOURCE_TREE STREQUAL LIBGIT2_EXPECTED_TREE)
  message(FATAL_ERROR "The libgit2 source tree does not match the required content digest.")
endif()
execute_process(COMMAND "${GIT_EXECUTABLE}" -C "${LIBGIT2_SOURCE_DIR}" status --porcelain --untracked-files=all
  OUTPUT_VARIABLE LIBGIT2_SOURCE_STATUS OUTPUT_STRIP_TRAILING_WHITESPACE
  RESULT_VARIABLE LIBGIT2_STATUS_RESULT)
if(NOT LIBGIT2_STATUS_RESULT EQUAL 0 OR NOT LIBGIT2_SOURCE_STATUS STREQUAL "")
  message(FATAL_ERROR "The libgit2 source tree must contain no local changes or untracked build inputs.")
endif()

set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_CLI OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_FUZZERS OFF CACHE BOOL "" FORCE)
set(USE_SSH OFF CACHE STRING "" FORCE)
set(USE_HTTPS OFF CACHE STRING "" FORCE)
set(USE_SHA1 CollisionDetection CACHE STRING "" FORCE)
set(USE_SHA256 Builtin CACHE STRING "" FORCE)
set(USE_HTTP_PARSER builtin CACHE STRING "" FORCE)
if(MSVC)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  set(STATIC_CRT ON CACHE BOOL "" FORCE)
endif()
add_subdirectory("${LIBGIT2_SOURCE_DIR}" "${CMAKE_BINARY_DIR}/libgit2" EXCLUDE_FROM_ALL)

add_library(binding MODULE src/binding.cc src/utf8-decoder.cc src/git-status-native.cc)
target_compile_features(binding PRIVATE cxx_std_17)
target_compile_definitions(binding PRIVATE NAPI_VERSION=8)
target_include_directories(binding PRIVATE "${NODE_INCLUDE_DIR}" "${LIBGIT2_SOURCE_DIR}/include" "${CMAKE_BINARY_DIR}/libgit2/gen_headers")
target_link_libraries(binding PRIVATE libgit2package)
set_target_properties(binding PROPERTIES PREFIX "" SUFFIX ".node")
if(WIN32)
  if(NOT EXISTS "${NODE_LIBRARY}")
    message(FATAL_ERROR "NODE_LIBRARY must name the matching preinstalled node.lib.")
  endif()
  target_link_libraries(binding PRIVATE "${NODE_LIBRARY}")
  target_sources(binding PRIVATE src/host-loader-win.cc src/windows-queries.cc src/windows-rename.cc src/network-bridge-win.cc src/binding-icons-win.cc src/icons-bridge-win.cc src/binding-transfer-win.cc src/file-transfer-win.cc src/clipboard-win.cc src/clipboard-ole-engine.cc src/binding-desktop-win.cc src/desktop-bridge-win.cc src/window-actions-win.cc src/shell-context-win.cc src/windows-async-work.cc)
  target_sources(binding PRIVATE src/locks-win.cc)
  target_link_libraries(binding PRIVATE delayimp mpr shell32 shlwapi gdiplus ole32 user32 uuid psapi advapi32)
  target_link_options(binding PRIVATE /DELAYLOAD:node.exe)
  target_link_options(binding PRIVATE /LTCG)
  target_compile_options(binding PRIVATE /W4 /WX /EHsc)
  set_property(TARGET binding PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
elseif(APPLE)
  target_link_options(binding PRIVATE "-undefined" "dynamic_lookup")
  target_compile_options(binding PRIVATE -Wall -Wextra -Werror)
else()
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(Threads REQUIRED)
    set(GLIB_SDK_ROOT "" CACHE PATH "Path to the preinstalled GLib development SDK prefix.")
    find_path(GLIB_INCLUDE_DIR gio/gio.h HINTS "${GLIB_SDK_ROOT}/include/glib-2.0" PATHS /usr/include/glib-2.0)
    find_path(GLIB_CONFIG_INCLUDE_DIR glibconfig.h HINTS "${GLIB_SDK_ROOT}/lib/${CMAKE_LIBRARY_ARCHITECTURE}/glib-2.0/include" PATHS "/usr/lib/${CMAKE_LIBRARY_ARCHITECTURE}/glib-2.0/include")
    if(NOT GLIB_INCLUDE_DIR OR NOT GLIB_CONFIG_INCLUDE_DIR)
      message(FATAL_ERROR "GLIB_SDK_ROOT must name the preinstalled GLib development SDK prefix.")
    endif()
    target_sources(binding PRIVATE src/locks-linux.cc src/binding-locks-linux.cc src/linux-work.cc)
    target_sources(binding PRIVATE src/desktop-linux.cc src/binding-desktop-linux.cc)
    target_include_directories(binding PRIVATE "${GLIB_INCLUDE_DIR}" "${GLIB_CONFIG_INCLUDE_DIR}")
    target_link_libraries(binding PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
  endif()
  target_compile_options(binding PRIVATE -Wall -Wextra -Werror)
endif()
install(TARGETS binding LIBRARY DESTINATION native RUNTIME DESTINATION native)
